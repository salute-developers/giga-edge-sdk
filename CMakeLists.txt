cmake_minimum_required(VERSION 3.24)
project(giga_edge_runtime)

set(CMAKE_CXX_STANDARD 17)

include(GNUInstallDirs)

list(APPEND CMAKE_MODULE_PATH "${PROJECT_SOURCE_DIR}/cmake/find")

if(CMAKE_PROJECT_NAME STREQUAL PROJECT_NAME)
  include(CTest)
endif()

option(COMPILE_WITH_ADDRESS_SANITIZER "Compile with address sanitizer." OFF)
option(COMPILE_WITH_THREAD_SANITIZER "Compile with thread sanitizer." OFF)
option(ARCHIVE_SUPPORT "Support reading from archive" ON)
option(COMPILE_WITH_EXTERNAL_YAMLCPP_PACKAGE "Target yaml-cpp is provided by the parent project: do not look for the package" OFF)
option(COMPILE_SHARED_RUNTIME "Build installable libgiga_edge_runtime shared library" OFF)
option(GIGA_EDGE_STRIP_NODE_DOCS "Leave node docs and source paths out of the binaries" OFF)
set(MODULES "" CACHE STRING "Modules list to build")
set(PRESET "" CACHE STRING "Preset to load")

include(cmake/Module.cmake)
include(cmake/Tests.cmake)
include(cmake/Lib.cmake)
include(cmake/SharedRuntime.cmake)

# PRESET is the name of a preset from cmake/presets/ or an absolute path to a .cmake file.
if (PRESET)
    if(IS_ABSOLUTE "${PRESET}")
        set(preset_file "${PRESET}")
    else()
        set(preset_file "cmake/presets/${PRESET}.cmake")
    endif()
    include(${preset_file} OPTIONAL RESULT_VARIABLE preset_loaded)
    if(NOT preset_loaded)
        message(WARNING "Unknown preset '${PRESET}', using default configuration")
    endif()
endif()
if(COMPILE_SHARED_RUNTIME)
  # The static libraries of the SDK are linked into the shared one. Set after the preset: it may enable the option.
  set(CMAKE_POSITION_INDEPENDENT_CODE ON)
endif()

string(REPLACE " " ";" MODULES "${MODULES}")
# Explicitly named modules are checked against ALL_MODULES; "*" and the default set are not.
set(_named_modules "")
if(NOT "${MODULES}" STREQUAL "*")
    set(_named_modules ${MODULES})
endif()

# Modules are the directories of giga_edge/modules/; subdirectories inherit ALL_MODULES and the module properties.
include(giga_edge/modules/modules.cmake)

if("${MODULES}" STREQUAL "*")
    set(MODULES ${ALL_MODULES})
endif()

if("${MODULES}" STREQUAL "")
    set(MODULES ${GIGA_EDGE_DEFAULT_MODULES})
endif()

foreach(_m ${_named_modules})
    if(NOT "${_m}" IN_LIST ALL_MODULES)
        message(FATAL_ERROR "Module '${_m}' is not in this distribution or is unknown")
    endif()
endforeach()

foreach(_m ${ALL_MODULES})
    foreach(_dep ${MODULE_DEPS_${_m}})
        if(NOT "${_dep}" IN_LIST ALL_MODULES)
            message(FATAL_ERROR "MODULE_DEPS_${_m} references '${_dep}' which is not in ALL_MODULES")
        endif()
    endforeach()
endforeach()

list(LENGTH ALL_MODULES _module_count)
foreach(_pass RANGE ${_module_count})
    set(_added_this_pass FALSE)
    foreach(_m ${MODULES})
        foreach(_dep ${MODULE_DEPS_${_m}})
            if(NOT "${_dep}" IN_LIST MODULES)
                list(APPEND MODULES ${_dep})
                set(_added_this_pass TRUE)
                message(STATUS "Auto-enabling module '${_dep}' (required by '${_m}')")
            endif()
        endforeach()
    endforeach()
    if(NOT _added_this_pass)
        break()
    endif()
endforeach()
list(REMOVE_DUPLICATES MODULES)


message(STATUS "Building modules: ${MODULES}")
set(_backends "TFLITE=${TFLITE_ENABLED} ONNXRUNTIME=${ONNXRUNTIME_ENABLED} LLAMA_CPP=${LLAMA_CPP_ENABLED}")
message(STATUS "Preset: '${PRESET}'  Backends: ${_backends}")

IF (COMPILE_WITH_ADDRESS_SANITIZER)
    IF (ANDROID)
        # There is no separate libpthread on Android: the functionality is all in libc.
        # https://developer.android.com/ndk/guides/stable_apis
        # https://cmake.org/cmake/help/v3.15/module/FindThreads.html
        set(CMAKE_THREAD_LIBS_INIT "-lc")
        set(CMAKE_HAVE_THREADS_LIBRARY 1)
        set(CMAKE_USE_WIN32_THREADS_INIT 0)
        set(CMAKE_USE_PTHREADS_INIT 1)
    ENDIF()

    set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer")
    set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -fsanitize=address,undefined")
ENDIF ()

IF (COMPILE_WITH_THREAD_SANITIZER)
    set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -fsanitize=thread -fno-omit-frame-pointer")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fsanitize=thread -fno-omit-frame-pointer")
    set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -fsanitize=thread")
ENDIF ()

include_directories(
    ${PROJECT_SOURCE_DIR}
    ${PROJECT_SOURCE_DIR}/giga_edge
    ${PROJECT_SOURCE_DIR}/giga_edge/nodes/include
    # all.h is generated by giga_edge/nodes/CMakeLists.txt.
    ${PROJECT_BINARY_DIR}/giga_edge/nodes/include
    ${PROJECT_SOURCE_DIR}/giga_edge/contrib
)

if(MSVC)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W3 /utf-8 /EHsc /bigobj /Zc:__cplusplus /Zc:preprocessor)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES EIGEN_HAS_STD_RESULT_OF=0)
else()
  add_compile_options(-Wall)
endif()
# The MLAS assembly of ORT is built without .note.GNU-stack, so the linker marks the module as requiring an executable
# stack; glibc 2.41+ refuses to load such a module ("cannot enable executable stack").
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_link_options(LINKER:-z,noexecstack)
endif()
if(GIGA_EDGE_STRIP_NODE_DOCS)
  add_compile_definitions(GIGA_EDGE_STRIP_NODE_DOCS)
endif()
include(cmake/Dependencies.cmake)

add_subdirectory(giga_edge)
